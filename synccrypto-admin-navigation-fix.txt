/*
  Add this block inside your existing <script type="module">, near the end of
  DOMContentLoaded, before its closing `});`. Do not add a second script module
  if you can insert it into the existing one.
*/
function setupAdminNavigation() {
  const sidebar = document.getElementById("adminSidebar");
  const toggle = document.getElementById("sidebarToggle");
  const overlay = document.getElementById("sidebarOverlay");
  const links = [...document.querySelectorAll("#adminSidebar .sidebar-link")];

  const sectionIds = {
    overview: "section-overview",
    users: "section-users",
    funding: "section-funding",
    withdrawals: "section-withdrawals",
    transactions: "section-transactions",
    notifications: "section-notifications",
    activity: "section-activity",
    settings: "section-settings"
  };

  function closeSidebar() {
    sidebar?.classList.remove("sidebar-open");
    overlay?.classList.remove("overlay-visible");
    if (toggle) toggle.setAttribute("aria-expanded", "false");
  }

  function openSidebar() {
    sidebar?.classList.add("sidebar-open");
    overlay?.classList.add("overlay-visible");
    if (toggle) toggle.setAttribute("aria-expanded", "true");
  }

  function activateSection(sectionName) {
    const targetId = sectionIds[sectionName];
    const target = targetId && document.getElementById(targetId);

    if (!target) {
      console.warn(`Admin section not found: ${sectionName}`);
      return;
    }

    document.querySelectorAll(".admin-content-section").forEach(section => {
      section.classList.remove("section-active");
    });
    target.classList.add("section-active");

    links.forEach(link => {
      const active = link.dataset.section === sectionName;
      link.classList.toggle("nav-active", active);
      link.classList.toggle("bg-yellow-400/10", active);
      link.classList.toggle("text-yellow-400", active);
      link.classList.toggle("text-slate-400", !active);
      link.setAttribute("aria-current", active ? "page" : "false");
    });

    closeSidebar();
    window.scrollTo({ top: 0, behavior: "smooth" });
  }

  toggle?.addEventListener("click", () => {
    if (sidebar?.classList.contains("sidebar-open")) closeSidebar();
    else openSidebar();
  });

  overlay?.addEventListener("click", closeSidebar);

  links.forEach(link => {
    link.addEventListener("click", event => {
      event.preventDefault();
      activateSection(link.dataset.section);
    });
  });

  document.addEventListener("keydown", event => {
    if (event.key === "Escape") closeSidebar();
  });

  // Start at the overview section.
  activateSection("overview");
}

// Call setupAdminNavigation() once inside DOMContentLoaded.
setupAdminNavigation();
</script>
